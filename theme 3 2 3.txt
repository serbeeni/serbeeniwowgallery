function interceptTumblrMice(e) {
    const lightbox = e.target.closest('.tmblr-lightbox, #tumblr_lightbox, .lightbox-wrapper');
    if (!lightbox || e.target.tagName !== 'IMG') return;
    
    if (e.target.closest('.tmblr-lightbox-close, .lightbox-close, .close-button, .tmblr-lightbox-next, .tmblr-lightbox-prev')) {
        return;
    }

    e.preventDefault();
    e.stopImmediatePropagation();
    if (e.type === 'click') {
        launchPerfectFullscreen(e.target);
    }
}
document.addEventListener('click', interceptTumblrMice, { capture: true });
document.addEventListener('mousedown', interceptTumblrMice, { capture: true });
document.addEventListener('mouseup', interceptTumblrMice, { capture: true });

function launchPerfectFullscreen(targetImg) {
    if (document.getElementById('custom-fullscreen-overlay')) return;

    const overlay = document.createElement('div');
    overlay.id = 'custom-fullscreen-overlay';
    overlay.style.cssText = `
        position: fixed !important;
        top: 0 !important;
        left: 0 !important;
        width: 100vw !important;
        height: 100vh !important;
        background: #000000 !important;
        z-index: 2147483647 !important;
        display: flex !important;
        align-items: center !important;
        justify-content: center !important;
        cursor: url('https://files.catbox.moe/pc0q9w.cur'), zoom-out !important;
    `;

    const fsImg = document.createElement('img');
    fsImg.src = targetImg.getAttribute('data-highres') || targetImg.getAttribute('data-src') || targetImg.src;
    fsImg.style.cssText = `
        width: 100% !important;
        height: 100% !important;
        max-width: 100vw !important;
        max-height: 100vh !important;
        margin: 0 !important;
        padding: 0 !important;
        display: block !important;
        object-fit: contain !important;
    `;

    overlay.appendChild(fsImg);
    document.body.appendChild(overlay);

    if (overlay.requestFullscreen) {
        overlay.requestFullscreen().catch(() => {});
    } else if (overlay.webkitRequestFullscreen) {
        overlay.webkitRequestFullscreen();
    }

    overlay.addEventListener('click', function(evt) {
        evt.preventDefault();
        evt.stopImmediatePropagation();
        triggerSafeClose();
    }, { capture: true });
}

function triggerSafeClose() {
    const overlay = document.getElementById('custom-fullscreen-overlay');
    if (!overlay) return;
    const isFullscreen = document.fullscreenElement === overlay || document.webkitFullscreenElement === overlay;
    if (isFullscreen) {
        if (document.exitFullscreen) document.exitFullscreen().catch(() => { overlay.remove(); });
        else if (document.webkitExitFullscreen) document.webkitExitFullscreen();
    } else {
        overlay.remove();
    }
}

document.addEventListener('keydown', function(evt) {
    if (evt.key === 'Escape') {
        const overlay = document.getElementById('custom-fullscreen-overlay');
        if (overlay) {
            evt.preventDefault();
            evt.stopImmediatePropagation();
            triggerSafeClose();
        }
    }
}, { capture: true });

function handleFullscreenChange(e) {
    const overlay = document.getElementById('custom-fullscreen-overlay');
    if (overlay) {
        e.stopImmediatePropagation();
        if (!document.fullscreenElement && !document.webkitFullscreenElement) {
            overlay.remove();
        }
    }
}
document.addEventListener('fullscreenchange', handleFullscreenChange, { capture: true });
document.addEventListener('webkitfullscreenchange', handleFullscreenChange, { capture: true });

function buildCaptionFilters() {
    const dropdown = document.getElementById('filterMenu');
    if (!dropdown) return;
    
    dropdown.innerHTML = '';

    if (isBatchLoading) {
        const waitItem = document.createElement('a');
        waitItem.textContent = "Please wait...";
        waitItem.href = "#";
        waitItem.style.cursor = "default";
        waitItem.style.opacity = "0.6";
        waitItem.onclick = (e) => e.preventDefault();
        dropdown.appendChild(waitItem);
        return;
    }

	if (!allLogsLoaded && nextPageUrl) {
        const loadItem = document.createElement('a');
        loadItem.textContent = "Load All";
        loadItem.href = "#";
        loadItem.onclick = (e) => {
            e.preventDefault();
            e.stopPropagation(); // STOPS EVENT FROM BUBBLING TO WINDOW & CLOSING MENU
            loadAllPosts();
        };
        dropdown.appendChild(loadItem);
        return;
    }
    const allLink = document.createElement('a');
    allLink.textContent = "All";
    allLink.href = "#";
    allLink.onclick = function(e) {
        e.preventDefault();
        filterPosts('all');
        closeAllMenus();
    };
    dropdown.appendChild(allLink);
    
    const posts = document.querySelectorAll('article.post');
    const locations = new Set();
    posts.forEach(function(post) {
        const match = post.textContent.match(/-\s*(.+)$/);
        if (match) {
            locations.add(match[1].trim());
        } else {
            const fallbackText = post.textContent.trim();
            if (fallbackText) locations.add(fallbackText);
        }
    });

    const sortedLocations = Array.from(locations).sort((a, b) => a.localeCompare(b, undefined, { sensitivity: 'base' }));

    const categoryOrder = ["Kalimdor", "Eastern Kingdoms", "Dungeons", "Other"];
    const categoryMap = { "Kalimdor": [], "Eastern Kingdoms": [], "Dungeons": [], "Other": [] };

    sortedLocations.forEach(function(loc) {
        const locKey = Object.keys(FILTER_DICTIONARY).find(k => k.toLowerCase() === loc.toLowerCase());
        const category = locKey ? FILTER_DICTIONARY[locKey] : "Other";
        categoryMap[category].push(loc);
    });

    categoryOrder.forEach(function(catName) {
        const catLocations = categoryMap[catName];
        if (!catLocations || catLocations.length === 0) return;

        const subContainer = document.createElement('div');
        subContainer.className = 'submenu-item';

        const titleDiv = document.createElement('div');
        titleDiv.className = 'submenu-title';

        const labelSpan = document.createElement('span');
        labelSpan.textContent = catName;

        const arrowSpan = document.createElement('span');
        arrowSpan.style.fontSize = '10px';
        arrowSpan.style.marginLeft = '10px';
        arrowSpan.textContent = '▶';

        titleDiv.appendChild(labelSpan);
        titleDiv.appendChild(arrowSpan);

        const subContent = document.createElement('div');
        subContent.className = 'submenu-content';

        catLocations.forEach(function(loc) {
            const link = document.createElement('a');
            link.textContent = loc.charAt(0).toUpperCase() + loc.slice(1);
            link.href = "#";
            link.onclick = function(e) { 
                e.preventDefault(); 
                filterPosts(loc); 
                closeAllMenus();
            };
            subContent.appendChild(link);
        });

        subContainer.appendChild(titleDiv);
        subContainer.appendChild(subContent);
        dropdown.appendChild(subContainer);
    });
}

function fetchNextPage() {
    if (!nextPageUrl || isFetchingPage) return Promise.resolve(false);
    isFetchingPage = true;
    scrollStatus.style.display = 'block';

    return fetch(nextPageUrl)
        .then(response => response.text())
        .then(htmlString => {
            const parser = new DOMParser();
            const doc = parser.parseFromString(htmlString, 'text/html');
            const fragment = doc.getElementById('contentContainer');
         
            if (fragment) {
                const newPosts = fragment.querySelectorAll('article.post');
                newPosts.forEach(post => {
                    generateGridThumbnails(fragment);
                    container.appendChild(post);

                    if (currentFilter !== 'all') {
                        const match = post.textContent.match(/-\s*(.+)$/);
                        const postLoc = match ? match[1].trim().toLowerCase() : post.textContent.trim().toLowerCase();
                        post.style.display = (postLoc === currentFilter.toLowerCase()) ? 'block' : 'none';
                    }
                });
            }
        
            const newNextLink = doc.getElementById('nextPageLink');
            nextPageUrl = newNextLink ? newNextLink.href : null;
            isFetchingPage = false;
            scrollStatus.style.display = 'none';
            buildCaptionFilters();
            return true;
        })
        .catch(err => {
            console.error("Infinite scroll failed:", err);
            isFetchingPage = false;
            scrollStatus.style.display = 'none';
            return false;
        });
}

async function loadAllPosts() {
    const btn = document.getElementById('loadAllBtn');
    const btnText = document.getElementById('loadAllBtnText');
    const track = document.getElementById('loadBarTrack');
    const fill = document.getElementById('loadBarFill');

    if (isBatchLoading) return;

    if (!nextPageUrl) {
        allLogsLoaded = true;
        if (btn) btn.disabled = true;
        if (btnText) btnText.innerText = "ALL LOADED";
        if (track && fill) {
            track.classList.add('active');
            fill.style.width = '100%';
            setTimeout(() => track.classList.remove('active'), 300);
        }
        buildCaptionFilters();
        return;
    }

    isBatchLoading = true;
    if (btn) btn.disabled = true;
    if (btnText) btnText.innerText = "LOADING...";
    
    let currentProgress = 15;
    if (track && fill) {
        track.classList.add('active');
        fill.style.width = currentProgress + '%';
    }

    buildCaptionFilters();

    while (nextPageUrl) {
        const success = await fetchNextPage();
        
        if (currentProgress < 90) {
            currentProgress += (90 - currentProgress) * 0.25;
            if (fill) fill.style.width = currentProgress + '%';
        }

        if (!success) {
            await new Promise(resolve => setTimeout(resolve, 200));
            if (!nextPageUrl) break;
        }
    }

    allLogsLoaded = true;
    isBatchLoading = false;

    if (btn) btn.disabled = true;
    if (btnText) btnText.innerText = "ALL LOADED";
    if (track && fill) {
        fill.style.width = '100%';
        setTimeout(() => {
            track.classList.remove('active');
        }, 300);
    }
    buildCaptionFilters();
}

function filterPosts(loc) {
    if (loc === currentFilter) return;
    currentFilter = loc;
    const posts = document.querySelectorAll('article.post');
    posts.forEach(post => {
        if (loc === 'all') {
            post.style.display = 'block';
        } else {
            const match = post.textContent.match(/-\s*(.+)$/);
            const postLoc = match ? match[1].trim().toLowerCase() : post.textContent.trim().toLowerCase();
            post.style.display = (postLoc === loc.toLowerCase()) ? 'block' : 'none';
        }
    });
}

document.addEventListener('keydown', function(e) {
    const activeTag = document.activeElement ? document.activeElement.tagName : '';
    if (['INPUT', 'TEXTAREA', 'SELECT'].includes(activeTag) || document.activeElement.isContentEditable) return;
    if (e.ctrlKey || e.altKey || e.metaKey) return;

    const code = e.code;
    const key = e.key ? e.key.toLowerCase() : '';

    if (code === 'KeyM' || key === 'm') {
        e.preventDefault();
        toggleMenu();
    } else if (code === 'KeyF' || key === 'f') {
        e.preventDefault();
        toggleFilterMenu();
    } else if (code === 'KeyL' || key === 'l') {
        e.preventDefault();
        loadAllPosts();
    } else if (code === 'KeyG' || key === 'g') {
        e.preventDefault();
        toggleLayout();
    }
}, true);

(function() {
    let isOldestFirst = false;

    // 1. Create the button dynamically
    const sortBtn = document.createElement('button');
    sortBtn.className = 'toggle-btn';
    sortBtn.id = 'standaloneSortBtn';
    sortBtn.innerText = 'Newest First';
    sortBtn.style.display = 'none';

    // 2. Inject it directly to the left of the Load All button
    function injectSortBtn() {
        const loadBtnWrapper = document.querySelector('.load-btn-wrapper') || document.getElementById('loadAllBtn');
        if (loadBtnWrapper && loadBtnWrapper.parentNode) {
            loadBtnWrapper.parentNode.insertBefore(sortBtn, loadBtnWrapper);
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', injectSortBtn);
    } else {
        injectSortBtn();
    }

    // 3. Sorting behavior
    sortBtn.onclick = function() {
        const container = document.getElementById('contentContainer');
        if (!container) return;

        const posts = Array.from(container.querySelectorAll('article.post'));
        isOldestFirst = !isOldestFirst;

        posts.reverse().forEach(post => container.appendChild(post));
        sortBtn.innerText = isOldestFirst ? "Oldest First" : "Newest First";
    };

    // 4. Background observer to reveal the button when "ALL LOADED" appears
    const observer = new MutationObserver(() => {
        const loadBtnText = document.getElementById('loadAllBtnText');
        const isFinished = (loadBtnText && loadBtnText.innerText.toUpperCase().includes('ALL LOADED')) || 
                           (typeof allLogsLoaded !== 'undefined' && allLogsLoaded === true);

        if (isFinished && sortBtn.style.display === 'none') {
            sortBtn.style.display = 'inline-block';
        }
    });

    observer.observe(document.body, { childList: true, subtree: true, characterData: true });

    // 5. Hotkey Listener ('S')
    document.addEventListener('keydown', function(e) {
        const activeTag = document.activeElement ? document.activeElement.tagName : '';
        if (['INPUT', 'TEXTAREA', 'SELECT'].includes(activeTag) || (document.activeElement && document.activeElement.isContentEditable)) {
            return;
        }

        if ((e.key === 's' || e.key === 'S') && !e.ctrlKey && !e.altKey && !e.metaKey) {
            if (sortBtn && sortBtn.style.display !== 'none') {
                e.preventDefault();
                sortBtn.click();
            }
        }
    });
})();

let ytPlayer;

function onYouTubeIframeAPIReady() {
    ytPlayer = new YT.Player('youtube-audio-player', {

        videoId: DARNASSUS_VIDEO_ID,
        playerVars: {
            'autoplay': 0,
            'controls': 0,
            'disablekb': 1
        },
        events: {

            'onStateChange': onPlayerStateChange
        }
    });

function updateTrackTitleDisplay() {
    const textSpan = document.getElementById('nowPlayingText');
    const trackBadge = document.getElementById('nowPlayingTrack');
    if (!textSpan || !trackBadge || !ytPlayer || typeof ytPlayer.getVideoData !== 'function') return;

    const data = ytPlayer.getVideoData();
    if (data && data.title && isPlaying) {
        textSpan.innerText = '♪ ' + data.title;
        trackBadge.classList.add('has-track');
    } else {
        trackBadge.classList.remove('has-track');
    }
}

function onPlayerStateChange(event) {
    if (event.data === YT.PlayerState.ENDED && isPlaying) {
        ytPlayer.seekTo(0);
        ytPlayer.playVideo();
    }

    if (event.data === YT.PlayerState.PLAYING) {
        updateTrackTitleDisplay();
    }
}

function playRandomPlaylistTrack() {
    const randomIndex = Math.floor(Math.random() * 50);
    ytPlayer.loadPlaylist({
        list: WOW_PLAYLIST_ID,
        listType: 'playlist',
        index: randomIndex
    });
}

window.changeVolume = function(val) {
    if (ytPlayer && typeof ytPlayer.setVolume === 'function') {

    }

window.toggleMusic = function() {
    const btn = document.getElementById('musicToggleBtn');
    const trackBadge = document.getElementById('nowPlayingTrack');
    if (!isPlayerReady) return;

    if (!isPlaying) {
        if (isFirstPlay) {
            ytPlayer.loadVideoById(DARNASSUS_VIDEO_ID);
            ytPlayer.playVideo();
            isFirstPlay = false;
        } else {
            playRandomPlaylistTrack();
        }
        isPlaying = true;
        if (btn) btn.innerHTML = "♪";
    } else {
        ytPlayer.pauseVideo();
        isPlaying = false;
        if (btn) btn.innerHTML = "<s>♪</s>";
        if (trackBadge) trackBadge.classList.remove('has-track');
    }
};
